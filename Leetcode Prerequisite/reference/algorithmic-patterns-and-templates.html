1. One-Pass Complement HashMap (O(n))

val seen = HashMap<Int, Int>() // val -> index
for (i in nums.indices) {
    val prev = seen[target - nums[i]]
    if (prev != null) return intArrayOf(prev, i)
    seen[nums[i]] = i // Insert AFTER check!
}

2. O(1) Prefix Sum Range Query (LongArray)

val prefix = LongArray(nums.size + 1)
for (i in nums.indices) {
    prefix[i + 1] = prefix[i] + nums[i]
}
// Sum of nums[l..r] inclusive in O(1):
val subSum = prefix[r + 1] - prefix[l]

3. Converging & Read/Write Two Pointers

// A) Converging on sorted array: while (l < r)
// B) In-place compaction (#283 Move Zeroes):
var write = 0
for (read in nums.indices) {
    if (nums[read] != 0) {
        val t = nums[write]; nums[write] = nums[read]; nums[read] = t
        write++
    }
}

4. Amortized O(n) Shrinking Sliding Window

var left = 0; var winSum = 0; var best = 0
for (right in nums.indices) {
    winSum += nums[right]
    while (winSum > limit) {
        winSum -= nums[left++] // Moves forward ≤ n times total!
    }
    best = maxOf(best, right - left + 1)
}

5. Null-Safe Matching / Monotonic Stack

val st = ArrayDeque<Char>()
for (ch in s) {
    if (ch in "([{") st.addLast(ch)
    else {
        val open = st.removeLastOrNull() ?: return false
        if (!matches(open, ch)) return false
    }
}
return st.isEmpty()

6. firstTrue Monotonic Boundary Binary Search

var left = minAns
var right = maxAns
while (left < right) {
    val mid = left + (right - left) / 2
    if (feasible(mid)) right = mid   // Keep valid mid!
    else left = mid + 1              // Discard invalid mid
}
return left

7. ListNode Dummy Head & In-Place Reversal

// Reverse list without losing tail (#206):
var prev: ListNode? = null; var curr = head
while (curr != null) {
    val nextTemp = curr.next // Save bridge FIRST!
    curr.next = prev
    prev = curr; curr = nextTemp
}
return prev

8. Tree DFS & Wave-Snapshot Level-Order BFS

val q = ArrayDeque<TreeNode>().apply { if (root != null) addLast(root) }
while (q.isNotEmpty()) {
    val levelSize = q.size // Freeze wave before pushing children!
    repeat(levelSize) {
        val u = q.removeFirst()
        u.left?.let(q::addLast); u.right?.let(q::addLast)
    }
}

10. Backtracking & Top-Down Memoization (DP)

// A) Backtrack: res.add(ArrayList(path)) at leaf; path.removeLast() on unwind
// B) Top-Down Memoization (O(2^n) -> O(n)):
val memo = IntArray(n + 1) { -1 }
fun dfs(k: Int): Int = when {
    k <= 1 -> 1
    memo[k] != -1 -> memo[k]
    else -> (dfs(k - 1) + dfs(k - 2)).also { memo[k] = it }
}
