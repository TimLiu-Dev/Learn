Spring Security 6 Filter Chain Architecture Demystified

Target: Master Spring Security 6 / Spring Boot 3 architecture. Understand SecurityFilterChain, DelegatingFilterProxy, stateless session management, and why CSRF is safely disabled on REST APIs.

📖 Primary Source Recommendation

Study the official authority: Spring Security Official Architecture Guide. Ground your learning in primary architecture rather than secondary summaries.

1. What Changed in Spring Security 6 / Spring Boot 3?

Legacy tutorials extend WebSecurityConfigurerAdapter. In Spring Boot 3 / Spring Security 6, this class is completely deleted! Modern security is configured via a component-based SecurityFilterChain bean using lambda DSL.

2. Stateless REST vs Stateful Web Sessions

  • Stateful Web Browsers: Store JSESSIONID cookies in server RAM, vulnerable to Cross-Site Request Forgery (CSRF). CSRF protection is mandatory.
  • Stateless REST & Mobile APIs: Mobile apps send an Authorization: Bearer header on every single call. Because mobile apps do not send automatic ambient browser cookies, CSRF is safely disabled (csrf.disable()) and session creation is set to STATELESS.

3. Production Dual Security Configuration

Modern SecurityFilterChain Bean 
package com.example.ecommerce.security;

import org.springframework.context.annotation.*;
import org.springframework.security.config.annotation.web.builders.HttpSecurity;
import org.springframework.security.config.annotation.web.configuration.EnableWebSecurity;
import org.springframework.security.config.http.SessionCreationPolicy;
import org.springframework.security.web.SecurityFilterChain;

@Configuration
@EnableWebSecurity
public class SecurityConfig {

    @Bean
    public SecurityFilterChain securityFilterChain(HttpSecurity http) throws Exception {
        return http
            // 1. Disable CSRF for stateless REST:
            .csrf(csrf -> csrf.disable())
            
            // 2. Stateless session policy:
            .sessionManagement(session -> session.sessionCreationPolicy(SessionCreationPolicy.STATELESS))
            
            // 3. Endpoint authorization rules:
            .authorizeHttpRequests(auth -> auth
                .requestMatchers("/api/v1/auth/**", "/swagger-ui/**", "/v3/api-docs/**").permitAll()
                .requestMatchers("/api/v1/admin/**").hasRole("ADMIN")
                .anyRequest().authenticated()
            )
            .build();
    }
}
package com.example.ecommerce.security

import org.springframework.context.annotation.*
import org.springframework.security.config.annotation.web.builders.HttpSecurity
import org.springframework.security.config.annotation.web.configuration.EnableWebSecurity
import org.springframework.security.config.http.SessionCreationPolicy
import org.springframework.security.web.SecurityFilterChain

@Configuration
@EnableWebSecurity
class SecurityConfig {

    @Bean
    fun securityFilterChain(http: HttpSecurity): SecurityFilterChain =
        http
            .csrf { it.disable() }
            .sessionManagement { it.sessionCreationPolicy(SessionCreationPolicy.STATELESS) }
            .authorizeHttpRequests { auth ->
                auth
                    .requestMatchers("/api/v1/auth/**", "/swagger-ui/**", "/v3/api-docs/**").permitAll()
                    .requestMatchers("/api/v1/admin/**").hasRole("ADMIN")
                    .anyRequest().authenticated()
            }
            .build()
}

package com.example.security;

import org.springframework.security.config.annotation.web.builders.HttpSecurity;
import org.springframework.security.config.http.SessionCreationPolicy;
import org.springframework.security.web.SecurityFilterChain;

public class SecurityHelper {
    public static SecurityFilterChain configureStateless(HttpSecurity http) throws Exception {
        return http
            .csrf(csrf -> csrf.disable())

            .build();
    }
}
