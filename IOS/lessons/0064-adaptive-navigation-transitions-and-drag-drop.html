🧠 The Senior Android Anchor: Systems Comparison

In Android, adaptive multi-pane UIs use Material 3 NavigationSuiteScaffold and ListDetailPaneScaffold, with AnimatedContent for transitions and dragAndDropSource/Target.

In modern SwiftUI, NavigationSplitView and the iOS 18+/26 Tab / TabSection API automatically morph between an iPhone bottom tab bar and an iPad/Mac floating sidebar. Meanwhile, ViewThatFits, the modern Transition protocol, and Transferable handle responsive layouts, GPU-accelerated transitions, and type-safe drag-and-drop.

struct SlideBlurTransition: Transition {
    func body(content: Content, phase: TransitionPhase) -> some View {
        content
            .opacity(phase.isIdentity ? 1 : 0)
            .offset(y: phase == .willAppear ? 16 : (phase == .didDisappear ? -16 : 0))
    }
}

⚠️ Senior Failure Mode & Anti-Pattern:

Hardcoding UIDevice.current.userInterfaceIdiom == .pad Checks: Checking device idiom breaks iPad Slide Over / Split View multitasking and Stage Manager, where an iPad app window has a .compact horizontal size class! Always branch layouts on @Environment(\.horizontalSizeClass) or ViewThatFits.

import Foundation

enum SizeClass { case compact, regular }
enum NavigationMode: Equatable { case compactStack, twoColumnSplit, threeColumnSplit }
enum TransitionPhaseSim { case willAppear, identity, didDisappear }

enum AdaptiveRouter {
    static func layoutMode(horizontalSizeClass: SizeClass, availableWidth: Double) -> NavigationMode {
        switch horizontalSizeClass {
        case .compact:
            return .compactStack
        case .regular:
            return availableWidth >= 900 ? .threeColumnSplit : .twoColumnSplit
        }
    }

    static func opacity(for phase: TransitionPhaseSim) -> Double {
        phase == .identity ? 1.0 : 0.0
    }
}

func runTests() {
    print("🧪 Running Lesson 9.4 Verification Tests...")
    assert(AdaptiveRouter.layoutMode(horizontalSizeClass: .compact, availableWidth: 390) == .compactStack)
    assert(AdaptiveRouter.layoutMode(horizontalSizeClass: .regular, availableWidth: 700) == .twoColumnSplit)
    assert(AdaptiveRouter.layoutMode(horizontalSizeClass: .regular, availableWidth: 1100) == .threeColumnSplit)
    assert(AdaptiveRouter.opacity(for: .willAppear) == 0.0)
    assert(AdaptiveRouter.opacity(for: .identity) == 1.0)
    assert(AdaptiveRouter.opacity(for: .didDisappear) == 0.0)
    print("  ✅ All tests passed!")
}
runTests()
