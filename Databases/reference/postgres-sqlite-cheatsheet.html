2. Essential Production Patterns

-- Safe Upsert + Read-After-Write
INSERT INTO inventory (sku, qty) VALUES ($1, $2)
ON CONFLICT (sku) DO UPDATE SET qty = inventory.qty + EXCLUDED.qty
RETURNING sku, qty;

-- Job Queue Claim (PostgreSQL — LIMIT 1 MUST precede FOR UPDATE SKIP LOCKED)
WITH next_job AS (
  SELECT id FROM jobs
  WHERE status = 'queued' AND run_at <= now()
  ORDER BY run_at, id
  LIMIT 1
  FOR UPDATE SKIP LOCKED
)
UPDATE jobs SET status = 'processing', locked_at = now(), lease_expires_at = now() + interval '30 seconds'
FROM next_job WHERE jobs.id = next_job.id
RETURNING jobs.*;

-- Zero-Downtime Migration Guard (PostgreSQL: low lock_timeout, unbounded online scan)
SET lock_timeout = '2s';
SET statement_timeout = '0';
CREATE INDEX CONCURRENTLY idx_orders_tenant_created ON orders (tenant_id, created_at DESC);

-- Tenant-Safe RLS View + InitPlan Policy (PostgreSQL 15+)
CREATE VIEW v_orders_safe WITH (security_invoker = true) AS
SELECT id, tenant_id, status FROM orders;

CREATE POLICY p_tenant ON orders
  USING (tenant_id = (SELECT NULLIF(current_setting('app.current_tenant_id', true), '')::uuid))
  WITH CHECK (tenant_id = (SELECT NULLIF(current_setting('app.current_tenant_id', true), '')::uuid));
