1. Core Financial Invariants

  • UUIDv7 time-ordered primary keys (B-tree locality!).
  • BIGINT minor units (amount_cents > 0).
  • ON DELETE RESTRICT on all ledger relations.
  • Zero-sum journal invariant: SUM(DEBIT) = SUM(CREDIT).

2. Concurrency & Reliability

  • UNIQUE(tenant_id, idempotency_key) stops double-charges.
  • Deterministic sorted account lock order prevents deadlocks.
  • FOR UPDATE SKIP LOCKED + partial index powers outbox workers.
  • PITR WAL archiving + automated restore drills (RPO < 1m).

3. Security, AI & Analytics

  • FORCE ROW LEVEL SECURITY + non-owner runtime role.
  • pgvector HNSW (halfvec) + tsvector GIN fused via RRF.
  • Debezium CDC WAL stream → Parquet/DuckDB daily volume audits.
  • pg_stat_statements + unused-index FinOps monitoring.

A. Ready-to-Deploy PostgreSQL 17 DDL + RLS + Hybrid Search + Outbox

-- PulseLedger PostgreSQL 17 Production Schema (excerpt for GitHub migrations/001_init.sql)
CREATE EXTENSION IF NOT EXISTS vector;

CREATE ROLE pulse_app NOINHERIT LOGIN PASSWORD 'replace_via_secrets_manager';

CREATE TABLE accounts (
  id            UUID PRIMARY KEY,                     -- Generated as UUIDv7 in app (or native uuidv7() in PG 18+)
  tenant_id     UUID NOT NULL,
  owner_ref     TEXT NOT NULL,
  currency      TEXT NOT NULL CHECK (currency IN ('USD', 'EUR', 'GBP')),
  account_type  TEXT NOT NULL CHECK (account_type IN ('ASSET', 'LIABILITY', 'REVENUE', 'FEE')),
  balance_cents BIGINT NOT NULL DEFAULT 0 CHECK (balance_cents >= 0),
  created_at    TIMESTAMPTZ NOT NULL DEFAULT now(),
  UNIQUE (tenant_id, owner_ref, currency),
  UNIQUE (tenant_id, id)
);

CREATE TABLE journal_entries (
  id              UUID PRIMARY KEY,
  tenant_id       UUID NOT NULL,
  idempotency_key TEXT NOT NULL,
  description     TEXT NOT NULL,
  created_at      TIMESTAMPTZ NOT NULL DEFAULT now(),
  UNIQUE (tenant_id, idempotency_key),
  UNIQUE (tenant_id, id)
);

CREATE TABLE ledger_lines (
  id           BIGINT GENERATED ALWAYS AS IDENTITY,
  tenant_id    UUID NOT NULL,
  entry_id     UUID NOT NULL,
  account_id   UUID NOT NULL,
  direction    TEXT NOT NULL CHECK (direction IN ('DEBIT', 'CREDIT')),
  amount_cents BIGINT NOT NULL CHECK (amount_cents > 0),
  created_at   TIMESTAMPTZ NOT NULL DEFAULT now(),
  PRIMARY KEY (created_at, id),
  FOREIGN KEY (tenant_id, entry_id)   REFERENCES journal_entries(tenant_id, id) ON DELETE RESTRICT,
  FOREIGN KEY (tenant_id, account_id) REFERENCES accounts(tenant_id, id)        ON DELETE RESTRICT
) PARTITION BY RANGE (created_at);

CREATE TABLE ledger_lines_2027_05 PARTITION OF ledger_lines
  FOR VALUES FROM ('2027-05-01 00:00:00+00') TO ('2027-06-01 00:00:00+00');

-- Deferred constraint trigger enforcing SUM(DEBIT) = SUM(CREDIT) at COMMIT:
CREATE OR REPLACE FUNCTION fn_verify_balanced_entry() RETURNS TRIGGER AS $$
DECLARE v_net BIGINT;
BEGIN
  SELECT COALESCE(SUM(CASE WHEN direction = 'DEBIT' THEN amount_cents ELSE -amount_cents END), 0)
    INTO v_net FROM ledger_lines WHERE entry_id = NEW.entry_id;
  IF v_net <> 0 THEN
    RAISE EXCEPTION 'Unbalanced journal entry % (net=%)', NEW.entry_id, v_net;
  END IF;
  RETURN NULL;
END;
$$ LANGUAGE plpgsql;

CREATE CONSTRAINT TRIGGER trg_verify_balanced_entry
  AFTER INSERT ON ledger_lines DEFERRABLE INITIALLY DEFERRED
  FOR EACH ROW EXECUTE FUNCTION fn_verify_balanced_entry();

CREATE TABLE outbox_events (
  id           BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  tenant_id    UUID NOT NULL,
  event_type   TEXT NOT NULL,
  payload      JSONB NOT NULL,
  status       TEXT NOT NULL DEFAULT 'PENDING' CHECK (status IN ('PENDING','RETRY','PROCESSING','DELIVERED','DEAD_LETTER')),
  attempts     INT NOT NULL DEFAULT 0,
  max_attempts INT NOT NULL DEFAULT 5,
  available_at TIMESTAMPTZ NOT NULL DEFAULT now()
);
CREATE INDEX idx_outbox_claimable ON outbox_events (available_at, id)
  WHERE status IN ('PENDING', 'RETRY');

-- Least-privilege grants + Multi-Tenant RLS (enabled + FORCED on all tenant tables!)
GRANT USAGE ON SCHEMA public TO pulse_app;
GRANT SELECT, INSERT, UPDATE ON accounts, journal_entries, ledger_lines, outbox_events TO pulse_app;

ALTER TABLE accounts        ENABLE ROW LEVEL SECURITY; ALTER TABLE accounts        FORCE ROW LEVEL SECURITY;
ALTER TABLE journal_entries ENABLE ROW LEVEL SECURITY; ALTER TABLE journal_entries FORCE ROW LEVEL SECURITY;
ALTER TABLE ledger_lines    ENABLE ROW LEVEL SECURITY; ALTER TABLE ledger_lines    FORCE ROW LEVEL SECURITY;

CREATE POLICY p_tenant_accounts ON accounts
  USING (tenant_id = (SELECT NULLIF(current_setting('app.current_tenant_id', true), '')::uuid))
  WITH CHECK (tenant_id = (SELECT NULLIF(current_setting('app.current_tenant_id', true), '')::uuid));
CREATE POLICY p_tenant_entries ON journal_entries
  USING (tenant_id = (SELECT NULLIF(current_setting('app.current_tenant_id', true), '')::uuid))
  WITH CHECK (tenant_id = (SELECT NULLIF(current_setting('app.current_tenant_id', true), '')::uuid));
CREATE POLICY p_tenant_lines ON ledger_lines
  USING (tenant_id = (SELECT NULLIF(current_setting('app.current_tenant_id', true), '')::uuid))
  WITH CHECK (tenant_id = (SELECT NULLIF(current_setting('app.current_tenant_id', true), '')::uuid));

-- Worker Claim Query (PostgreSQL: LIMIT 1 MUST precede FOR UPDATE SKIP LOCKED!):
WITH next_evt AS (
  SELECT id FROM outbox_events
  WHERE status IN ('PENDING', 'RETRY') AND available_at <= now() AND attempts < max_attempts
  ORDER BY available_at ASC, id ASC
  LIMIT 1
  FOR UPDATE SKIP LOCKED
)
UPDATE outbox_events o
SET status = 'PROCESSING', attempts = attempts + 1
FROM next_evt WHERE o.id = next_evt.id
RETURNING o.id, o.tenant_id, o.event_type, o.payload;

-- Hybrid Dispute Search (pgvector halfvec HNSW + tsvector GIN + RRF)
CREATE TABLE dispute_chunks (
  id         BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  tenant_id  UUID NOT NULL,
  dispute_id UUID NOT NULL,
  body       TEXT NOT NULL,
  tsv        TSVECTOR GENERATED ALWAYS AS (to_tsvector('english', body)) STORED,
  embedding  HALFVEC(1536) NOT NULL
);
CREATE INDEX idx_disputes_tsv  ON dispute_chunks USING GIN (tsv);
CREATE INDEX idx_disputes_hnsw ON dispute_chunks USING hnsw (embedding halfvec_cosine_ops) WITH (m = 16, ef_construction = 64);

pool_mode = transaction
max_client_conn = 5000

reserve_pool_size = 10
server_idle_timeout = 60
# App rule: NEVER use session SET; ALWAYS use SET LOCAL inside BEGIN..COMMIT!
