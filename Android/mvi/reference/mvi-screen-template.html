MVI Screen Template

The skeleton from Lesson 5, with the names made generic. Copy it, rename it, and fill it in.

1. Contract

data class XState(
    val input: String = "",                  // fields that coexist → data class
    val status: XStatus = XStatus.Idle,      // states that exclude each other → sealed
) {
    val canSubmit: Boolean get() = input.isNotBlank() && status != XStatus.InFlight
}
sealed interface XStatus { object Idle : XStatus; object InFlight : XStatus
                           data class Failed(val message: String) : XStatus; object Done : XStatus }

sealed interface XIntent { data class InputChanged(val value: String) : XIntent; object Submit : XIntent }
sealed interface XResult { data class InputUpdated(val value: String) : XResult
                           object Started : XResult; object Succeeded : XResult
                           data class Failed(val message: String) : XResult }

2. Reducer (pure, tested first)

fun reduce(state: XState, result: XResult): XState = when (result) {
    is XResult.InputUpdated -> state.copy(input = result.value)
    XResult.Started         -> state.copy(status = XStatus.InFlight)
    XResult.Succeeded       -> state.copy(status = XStatus.Done)
    is XResult.Failed       -> state.copy(status = XStatus.Failed(result.message))
}

3. ViewModel (one entry point)

class XViewModel(private val repo: XRepository) : ViewModel() {
    private val _state = MutableStateFlow(XState())
    val state: StateFlow<XState> = _state.asStateFlow()

    fun onIntent(intent: XIntent) = when (intent) {
        is XIntent.InputChanged -> dispatch(XResult.InputUpdated(intent.value))
        XIntent.Submit          -> submit()
    }

    private fun submit() {
        val current = _state.value
        if (!current.canSubmit) return            // guard: reduce InFlight synchronously
        dispatch(XResult.Started)
        viewModelScope.launch {
            dispatch(try { repo.send(current.input); XResult.Succeeded }
                     catch (e: XException) { XResult.Failed(e.message ?: "Failed") })
        }
    }

    private fun dispatch(result: XResult) = _state.update { reduce(it, result) }
}

4. Compose (route + stateless screen)

@Composable
fun XRoute(viewModel: XViewModel, onDone: () -> Unit) {
    val state by viewModel.state.collectAsStateWithLifecycle()
    val currentOnDone by rememberUpdatedState(onDone)
    LaunchedEffect(state.status) { if (state.status == XStatus.Done) currentOnDone() }
    XScreen(state, viewModel::onIntent)
}

Checklist

  • Can the State represent an impossible combination? If so, make that part sealed.
  • Is every transition covered by a reducer test?
  • Does rotation keep the data? Does a double-tap send only one request?
  • Are one-off events in State (if losing one would be a bug) or in a Channel (if it's cosmetic)? See the One-Off Events sheet.
  • Process death: are small inputs saved in SavedStateHandle?
  • Do you catch specific exceptions, not Exception? A broad catch swallows cancellation.
