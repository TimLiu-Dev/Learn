4. Reactive Flow Transformation Pipeline

queryFlow
    .debounce(300L)              // Wait for user typing pauses
    .map { it.trim() }           // Normalize text
    .distinctUntilChanged()      // Avoid redundant queries
    .filter { it.length >= 2 }   // Validation guard
    .flatMapLatest { query ->   // Cancel in-flight stale searches!
        repository.searchStream(query)
    }
    .flowOn(Dispatchers.IO)      // Upstream runs on IO
    .catch { emit(emptyList()) } // Isolate stream errors
