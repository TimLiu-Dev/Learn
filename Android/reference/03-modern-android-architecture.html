AnnotationComponentLifetime Scope

2. Single Source of Truth (SSOT) Flow

// The Offline-First Repository Pattern
class OfflineFirstRepository @Inject constructor(
    private val localDao: ItemDao,
    private val remoteApi: ItemApi
) {
    // UI continuously observes the local database
    val itemsStream: Flow<List<Item>> = localDao.observeAll()

    // Sync worker or refresh trigger updates database; UI receives update automatically
    suspend fun sync() {
        val networkData = remoteApi.fetchItems()
        localDao.upsertAll(networkData.map { it.toEntity() })
    }
}

4. Multi-Module Dependency Rules

:app
  └── Depends on all :feature:* modules & :core:*

:feature:<name>
  └── Depends ONLY on :core:data, :core:model, :core:designsystem
  └── NEVER depends on other :feature modules!

:core:data
  └── Depends on :core:database, :core:network, :core:model

:core:model
  └── Pure Kotlin module (ZERO Android SDK dependencies)
