4. Canonical Algorithmic Patterns in Idiomatic Kotlin

Pattern 1: Two Pointers (Opposite Ends)

fun twoSumSorted(numbers: IntArray, target: Int): IntArray {
    var left = 0
    var right = numbers.size - 1
    while (left < right) {
        val sum = numbers[left] + numbers[right]
        when {
            sum == target -> return intArrayOf(left + 1, right + 1)
            sum < target -> left++
            else -> right--
        }
    }
    return intArrayOf()
}

Pattern 2: Sliding Window (Dynamic Expansion & Contraction)

fun minSubArrayLen(target: Int, nums: IntArray): Int {
    var left = 0
    var currentSum = 0
    var minLen = Int.MAX_VALUE

    for (right in nums.indices) {
        currentSum += nums[right]
        while (currentSum >= target) {
            minLen = minOf(minLen, right - left + 1)
            currentSum -= nums[left]
            left++
        }
    }
    return if (minLen == Int.MAX_VALUE) 0 else minLen
}

Pattern 3: Monotonic Stack (Next Greater Element)

fun nextGreaterElement(nums: IntArray): IntArray {
    val result = IntArray(nums.size) { -1 }
    val stack = ArrayDeque<Int>() // stores indices

    for (i in nums.indices) {
        while (stack.isNotEmpty() && nums[stack.last()] < nums[i]) {
            val prevIdx = stack.removeLast()
            result[prevIdx] = nums[i]
        }
        stack.addLast(i)
    }
    return result
}

Pattern 4: Binary Search Bisect (Lower Bound / Search Space Reduction)

// Lower bound: finds first index where condition(index) is true
fun binarySearchLowerBound(low: Int, high: Int, predicate: (Int) -> Boolean): Int {
    var l = low
    var r = high
    while (l < r) {
        val mid = (l + r) ushr 1 // safe from overflow
        if (predicate(mid)) {
            r = mid
        } else {
            l = mid + 1
        }
    }
    return l
}

Pattern 5: Breadth-First Search (2D Matrix Island Traversal)

fun numIslands(grid: Array<CharArray>): Int {
    if (grid.isEmpty()) return 0
    val rows = grid.size
    val cols = grid[0].size
    var count = 0
    val queue = ArrayDeque<Pair<Int, Int>>()
    val dirs = arrayOf(0 to 1, 1 to 0, 0 to -1, -1 to 0)

    for (r in 0 until rows) {
        for (c in 0 until cols) {
            if (grid[r][c] == '1') {
                count++
                grid[r][c] = '0' // mark visited in-place
                queue.addLast(r to c)

                while (queue.isNotEmpty()) {
                    val (cr, cc) = queue.removeFirst()
                    for ((dr, dc) in dirs) {
                        val nr = cr + dr
                        val nc = cc + dc
                        if (nr in 0 until rows && nc in 0 until cols && grid[nr][nc] == '1') {
                            grid[nr][nc] = '0'
                            queue.addLast(nr to nc)
                        }
                    }
                }
            }
        }
    }
    return count
}

Pattern 6: Disjoint Set Union (Union-Find with Path Compression)

class UnionFind(n: Int) {
    private val parent = IntArray(n) { it }
    private val rank = IntArray(n) { 1 }

    fun find(x: Int): Int {
        if (parent[x] != x) {
            parent[x] = find(parent[x]) // path compression
        }
        return parent[x]
    }

    fun union(x: Int, y: Int): Boolean {
        val rootX = find(x)
        val rootY = find(y)
        if (rootX == rootY) return false // already connected

        // Union by rank
        if (rank[rootX] < rank[rootY]) {
            parent[rootX] = rootY
        } else if (rank[rootX] > rank[rootY]) {
            parent[rootY] = rootX
        } else {
            parent[rootY] = rootX
            rank[rootX]++
        }
        return true
    }
}
