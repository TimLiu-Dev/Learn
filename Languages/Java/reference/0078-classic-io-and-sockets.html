Classic java.io Decorators & Blocking Socket Rules

Quick checklist for choosing byte vs character streams, explicit UTF-8, flushing, and socket timeouts. 

// Safe Blocking Server Loop with Java 25 Virtual Threads
try (var server = new ServerSocket(9090)) {
    while (!Thread.currentThread().isInterrupted()) {
        Socket client = server.accept();
        Thread.ofVirtual().start(() -> handleClient(client));
    }
}

void handleClient(Socket socket) {
    try (socket;
         var in = new BufferedReader(new InputStreamReader(socket.getInputStream(), StandardCharsets.UTF_8));
         var out = new BufferedWriter(new OutputStreamWriter(socket.getOutputStream(), StandardCharsets.UTF_8))) {
        socket.setSoTimeout(3_000);
        String line = in.readLine();
        if (line != null && line.length() <= 4096) {
            out.write("ACK:" + line + "\n");
            out.flush();
        }
    } catch (IOException ignored) { /* log bounded warning */ }
}
