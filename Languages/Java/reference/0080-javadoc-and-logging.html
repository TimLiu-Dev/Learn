JEP 467 Markdown Javadoc & SLF4J / System.Logger

Quick reference for modern Javadoc syntax and zero-overhead production logging.

/// Summarizes payment batch using Markdown Javadoc (JEP 467).
/// - Validates non-empty payload
/// - Bridges cleanly through [org.slf4j.Logger] or [System.Logger]
/// @param id batch identifier
/// @return sanitized log tag
String formatTag(String id) {
    // Strip CR/LF before logging untrusted user strings!
    String clean = id.replace("\r", "_").replace("\n", "_");
    log.atDebug().addArgument(() -> expensiveJson()).setMessage("detail={}").log();
    return clean;
}
