01 Idempotency & Stale-Response Protection

@Test
fun `retried payment reuses identical Idempotency-Key across timeout recovery`() = runTest {
  server.enqueue(MockResponse().setSocketPolicy(SocketPolicy.NO_RESPONSE))
  server.enqueue(MockResponse().setResponseCode(200).setBody("""{"orderId":"ord-1"}"""))

  repo.payWithRetry(cartId = "c-9", idempotencyKey = "idem-123")

  val firstKey = server.takeRequest().getHeader("Idempotency-Key")
  val secondKey = server.takeRequest().getHeader("Idempotency-Key")
  assertThat(firstKey).isEqualTo("idem-123")
  assertThat(secondKey).isEqualTo("idem-123")
}
