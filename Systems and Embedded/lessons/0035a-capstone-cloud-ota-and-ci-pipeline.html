func IngestFrame(raw []byte, seen map[uint8]bool) (float32, error) {
    if len(raw) != 16 || raw[0] != 0xAA || raw[1] != 0x55 {
        return 0, errors.New("invalid sync framing")
    }
    seq := raw[2]
    if seen[seq] {
        return 0, nil // Idempotent ignore for QoS 1 redelivery
    }
    seen[seq] = true
    speedScaled := (uint16(raw[4]) << 8) | uint16(raw[5])
    return float32(speedScaled) / 100.0, nil
}

func IngestTelemetry(pkt []byte, seen map[uint8]bool) (float32, bool, error) {
    if len(pkt) != 16 || pkt[0] != 0xAA || pkt[1] != 0x55 {
        return 0, false, errors.New("bad packet")
    }
    seq := pkt[2]
    if seen[seq] {
        return 0, false, nil
    }
    seen[seq] = true
    raw := (uint16(pkt[4]) << 8) | uint16(pkt[5])
    return float32(raw) / 100.0, true, nil
}
