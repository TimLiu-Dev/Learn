Grand Capstone Part 2:
Zephyr RTOS MCU Firmware & Linux SPI Kernel Driver

In this second milestone of the Grand Capstone, you will implement the two lower tiers of our automotive system: the bare-metal Zephyr RTOS firmware running on the safety gateway microcontroller, and the Linux kernel character driver that receives high-speed SPI telemetry via DMA and hardware interrupts.

Mental Model Shift: User Space Polling vs Hardware Interrupt Driven DMA

In high-level application code, developers frequently read data using busy polling loops (e.g., while(true) { checkData(); sleep(10); }), wasting CPU cycles and introducing milliseconds of random latency.

In Systems Engineering, every byte transfer is interrupt-driven: when the Zephyr MCU finishes assembling a 16-byte packet, it toggles a hardware GPIO line. The Linux kernel's top-half ISR fires in nanoseconds, schedules a bottom-half threaded IRQ that triggers an SPI DMA read, and places the data directly into a wait-queue. The user space HAL daemon sleeps with zero CPU usage until woken!

1. Zephyr RTOS MCU Firmware Implementation

The microcontroller runs Zephyr RTOS on an STM32F4 cortex-M4 core. It receives CAN bus frames and broadcasts telemetry packets over high-speed SPI:

firmware/src/main.cZephyr RTOS C
#include <zephyr/kernel.h>
#include <zephyr/drivers/can.h>
#include <zephyr/drivers/spi.h>
#include <zephyr/drivers/gpio.h>

#define SPI_OP (SPI_OP_MODE_MASTER | SPI_WORD_SET(8) | SPI_TRANSFER_MSB)

const struct device *can_dev = DEVICE_DT_GET(DT_CHOSEN(zephyr_canbus));
const struct device *spi_dev = DEVICE_DT_GET(DT_NODELABEL(spi1));
static const struct gpio_dt_spec drdy_gpio = GPIO_DT_SPEC_GET(DT_NODELABEL(drdy_pin), gpios);

static uint8_t tx_packet[16];
static uint8_t seq_counter = 0;

void transmit_cockpit_packet(uint16_t speed_scaled, uint16_t rpm) {
    // 1. Pack framing and telemetry fields
    tx_packet[0] = 0xAA;
    tx_packet[1] = 0x55;
    tx_packet[2] = seq_counter++;
    tx_packet[3] = 0x01; // Dynamics packet
    tx_packet[4] = (speed_scaled >> 8) & 0xFF;
    tx_packet[5] = speed_scaled & 0xFF;
    tx_packet[6] = (rpm >> 8) & 0xFF;
    tx_packet[7] = rpm & 0xFF;
    tx_packet[8] = 0x04; // Drive (D)
    tx_packet[9] = 38;   // Battery temp: 38°C
    memset(&tx_packet[10], 0, 4);

    // 2. Compute CRC-16
    uint16_t crc = compute_crc16(tx_packet, 14);
    tx_packet[14] = (crc >> 8) & 0xFF;
    tx_packet[15] = crc & 0xFF;

    // 3. Pulse Data Ready (DRDY) GPIO line to wake Linux host
    gpio_pin_set_dt(&drdy_gpio, 1);

    // 4. Transmit over SPI
    const struct spi_buf buf = { .buf = tx_packet, .len = sizeof(tx_packet) };
    const struct spi_buf_set tx = { .buffers = &buf, .count = 1 };
    spi_write(spi_dev, &spi_cfg, &tx);

    gpio_pin_set_dt(&drdy_gpio, 0);
}

2. Linux Kernel SPI Slave Character Device Driver

On the application processor (e.g. NXP i.MX8 / QEMU ARM64 running Linux Kernel 6.6), the driver binds to the device tree node and creates /dev/cockpit_bridge:

drivers/misc/cockpit_bridge.cLinux Kernel C Driver
#include <linux/module.h>
#include <linux/spi/spi.h>
#include <linux/interrupt.h>
#include <linux/fs.h>
#include <linux/uaccess.h>
#include <linux/poll.h>

#define DEVICE_NAME "cockpit_bridge"
#define PACKET_SIZE 16

struct cockpit_dev {
    struct spi_device *spi;
    int irq_gpio;
    wait_queue_head_t wq;
    bool data_ready;
    uint8_t rx_buffer[PACKET_SIZE];
    struct mutex lock;
};

static struct cockpit_dev *g_dev;

// Threaded IRQ bottom-half: runs in process context when DRDY pin goes high
static irqreturn_t cockpit_threaded_irq(int irq, void *dev_id) {
    struct cockpit_dev *dev = dev_id;
    int ret;

    mutex_lock(&dev->lock);
    ret = spi_read(dev->spi, dev->rx_buffer, PACKET_SIZE);
    if (ret == 0) {
        dev->data_ready = true;
        wake_up_interruptible(&dev->wq);
    }
    mutex_unlock(&dev->lock);

    return IRQ_HANDLED;
}

// Read operation: copies telemetry packet to user space HAL
static ssize_t cockpit_read(struct file *filp, char __user *buf, size_t count, loff_t *f_pos) {
    struct cockpit_dev *dev = filp->private_data;
    int ret;

    if (count < PACKET_SIZE) return -EINVAL;

    // Sleep until threaded IRQ signals new packet arrival
    ret = wait_event_interruptible(dev->wq, dev->data_ready);
    if (ret) return -ERESTARTSYS;

    mutex_lock(&dev->lock);
    if (copy_to_user(buf, dev->rx_buffer, PACKET_SIZE)) {
        mutex_unlock(&dev->lock);
        return -EFAULT;
    }
    dev->data_ready = false;
    mutex_unlock(&dev->lock);

    return PACKET_SIZE;
}

static __poll_t cockpit_poll(struct file *filp, struct poll_table_struct *wait) {
    struct cockpit_dev *dev = filp->private_data;
    __poll_t mask = 0;

    poll_wait(filp, &dev->wq, wait);
    if (dev->data_ready) mask |= (EPOLLIN | EPOLLRDNORM);

    return mask;
}

static const struct file_operations cockpit_fops = {
    .owner   = THIS_MODULE,
    .read    = cockpit_read,
    .poll    = cockpit_poll,
};

Knowledge Retrieval & Systems Evaluation

Test your understanding of firmware-to-kernel communication and threaded interrupt handlers. All options are crafted to equal length.

1. Why does the Linux cockpit SPI driver use request_threaded_irq rather than performing spi_read inside a top-half ISR?
Top half interrupt handlers operate strictly in user space where hardware registers remain inaccessible.
SPI bus transfers require thread sleep context which is prohibited inside atomic ISR handlers.
Microcontroller hardware cannot trigger standard electrical GPIO transitions without active high speed network connections.
Linux compilers automatically delete top half interrupt routines to maximize operating system boot speeds.
2. What engineering benefit does supporting the poll file operation provide to the downstream user space HAL daemon?
It forces the graphics card to recalculate vehicle speed values using floating point hardware.
It grants the application root administrator privileges to modify arbitrary system kernel runtime settings.
It allows the daemon to sleep efficiently until hardware packets arrive without busy polling.
It encrypts telemetry payloads across flash memory partitions to satisfy local government privacy laws.
