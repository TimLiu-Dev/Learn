4. Rust Binder Server Skeleton libbinder_rs 
use binder::{BinderFeatures, Interface, ProcessState};

struct TelemetryService;
impl Interface for TelemetryService {}
impl ITelemetry for TelemetryService {
    fn getSpeed(&self) -> binder::Result<f32> {
        Ok(65.4)
    }
}

fn main() {
    let service = TelemetryService;
    let binder = BnTelemetry::new_binder(service, BinderFeatures::default());
    binder::add_service("android.hardware.telemetry.ITelemetry/default", binder.as_binder())
        .expect("Registration failed");
    ProcessState::join_thread_pool();
}
5. Linux Driver Threaded IRQ & Poll Kernel 6.6 
// Top-half: fast ACK, return IRQ_WAKE_THREAD
static irqreturn_t spi_top_half(int irq, void *dev_id) {
    return IRQ_WAKE_THREAD;
}

// Bottom-half: process context (can sleep)
static irqreturn_t spi_bottom_half(int irq, void *dev_id) {
    spi_sync(spi_dev, &msg);
    dev->data_ready = true;
    wake_up_interruptible(&dev->wq);
    return IRQ_HANDLED;
}

// User-space poll() handler
static __poll_t dev_poll(struct file *f, poll_table *wait) {
    poll_wait(f, &dev->wq, wait);
    return dev->data_ready ? (EPOLLIN | EPOLLRDNORM) : 0;
}
