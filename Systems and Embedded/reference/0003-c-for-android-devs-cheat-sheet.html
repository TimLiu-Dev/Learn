4. Structs, Alignment & Unaligned Parsing Lesson 0004c

Rule: Order fields largest alignment → smallest (uint64_t → uint32_t → uint16_t → uint8_t) to minimize silent struct padding. Enforce ABI layout at compile time with _Static_assert + offsetof(). 

/* NEVER do *(uint32_t *)(rx_buf + 1) on raw wire bytes! (UB/fault) */
static inline uint32_t read_le32(const uint8_t *p) {
    uint32_t v;
    memcpy(&v, p, sizeof(v)); /* Compiles to single unaligned load */
    return v;
}
_Static_assert(sizeof(struct CanHeader) == 8, "ABI size mismatch");

6. Safe Macros, UB Sanitizers & Makefile Lesson 0004d 
/* Wrap args AND expression in parens; use do/while(0) for stmts */
#define BIT_MASK(n)      (1u << (n))
#define REG_CLEAR(r, m)  do { (r) &= ~(m); } while (0)

# Strict Linux/Host Test Flags (catch UB & OOB immediately):
CFLAGS := -std=c11 -O2 -g -Wall -Wextra -Werror -Wconversion \
          -Wshadow -fsanitize=address,undefined

%.o: %.c
	$(CC) $(CFLAGS) -MMD -MP -c $< -o $@

Note: Always avoid signed integer overflow, shifting into the sign bit (1 << 31 on signed int), out-of-bounds pointer arithmetic, and strict-aliasing casts.
