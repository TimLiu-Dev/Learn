KMP Gradle Anatomy & Version Catalogs

Compressed reference sheet for Kotlin 2.x multiplatform build configuration and Gradle Version Catalogs (libs.versions.toml).

1. Canonical gradle/libs.versions.toml

[versions]
kotlin = "2.1.0"
agp = "8.7.2"
coroutines = "1.9.0"
ktor = "3.0.1"

[libraries]
kotlin-test = { module = "org.jetbrains.kotlin:kotlin-test", version.ref = "kotlin" }
coroutines-core = { module = "org.jetbrains.kotlinx:kotlinx-coroutines-core", version.ref = "coroutines" }
ktor-client-core = { module = "io.ktor:ktor-client-core", version.ref = "ktor" }
ktor-client-darwin = { module = "io.ktor:ktor-client-darwin", version.ref = "ktor" }
ktor-client-okhttp = { module = "io.ktor:ktor-client-okhttp", version.ref = "ktor" }

[plugins]
kotlin-multiplatform = { id = "org.jetbrains.kotlin.multiplatform", version.ref = "kotlin" }
android-library = { id = "com.android.library", version.ref = "agp" }

2. Modern build.gradle.kts Source Set Layout

plugins {
    alias(libs.plugins.kotlin.multiplatform)
    alias(libs.plugins.android.library)
}

kotlin {
    // 1. Android Target
    androidTarget {
        compilerOptions {
            jvmTarget.set(org.jetbrains.kotlin.gradle.dsl.JvmTarget.JVM_17)
        }
    }

    // 2. iOS Targets (Arm64 device + Simulator)
    listOf(
        iosArm64(),
        iosSimulatorArm64()
    ).forEach { iosTarget ->
        iosTarget.binaries.framework {
            baseName = "SharedAppCore"
            isStatic = true
        }
    }

    // 3. Hierarchical Source Sets
    sourceSets {
        commonMain.dependencies {
            implementation(libs.coroutines.core)
            implementation(libs.ktor.client.core)
        }
        commonTest.dependencies {
            implementation(libs.kotlin.test)
        }
        androidMain.dependencies {
            implementation(libs.ktor.client.okhttp)
        }
        iosMain.dependencies {
            implementation(libs.ktor.client.darwin)
        }
    }
}

android {
    namespace = "com.pulsesync.shared"
    compileSdk = 35
    defaultConfig { minSdk = 26 }
}

3. Source Set Hierarchy Diagram

              [ commonMain ]  <-- Shared logic, Ktor Core, Coroutines Core
               /          \
              /            \
       [ androidMain ]   [ iosMain ]  <-- Apple common target (Darwin engine)
                           /     \
                          /       \
                 [ iosArm64 ]   [ iosSimulatorArm64 ]

4. Key CLI Commands

  • ./gradlew tasks --group="verification" : List all testing tasks across JVM and Native.
  • ./gradlew check : Runs JVM unit tests and Apple target native unit tests.
  • ./gradlew linkSharedAppCoreDebugFrameworkIosSimulatorArm64 : Compiles iOS simulator framework.
