03 // PRODUCTION STARTER CODE (ARM64 + C)

Core Boot, Trap Vector & Syscall Gate Templates

arch/arm64/vectors.S — Exception Vector Entry & Trap Frame 
.macro SAVE_REGS
  sub sp, sp, #272
  stp x0,  x1,  [sp, #16 * 0]
  stp x2,  x3,  [sp, #16 * 1]
  // ... save x4..x29 ...
  mrs x21, elr_el1
  mrs x22, spsr_el1
  stp x30, x21, [sp, #16 * 15]
  str x22,      [sp, #16 * 16]
.endm

.align 11                 // VBAR_EL1 must be 2KB aligned
.global vectors
vectors:
  // Current EL with SP0 / SPx entries ...
  .balign 0x80
  b unhandled_exc
  // Lower EL using AArch64 (EL0 -> EL1 Synchronous Trap: SVC #0 / Page Fault)
  .org vectors + 0x400
  SAVE_REGS
  mov x0, sp              // Pass struct pt_regs* to C handler
  bl  el0_sync_handler
  b   restore_and_eret
kernel/trap.c — Syscall & Data Abort Dispatcher 
void el0_sync_handler(struct pt_regs *regs) {
  uint64_t esr;
  __asm__ volatile("mrs %0, esr_el1" : "=r"(esr));
  uint32_t ec = (esr >> 26) & 0x3F;

  if (ec == 0x15) { // SVC instruction from AArch64 EL0
    uint64_t nr = regs->regs[8];
    if (!seccomp_allowed(current_task(), nr)) {
      regs->regs[0] = (uint64_t)-EPERM;
      return;
    }
    regs->regs[0] = dispatch_syscall(nr, regs->regs[0], regs->regs[1], regs->regs[2]);
  } else if (ec == 0x24) { // Data Abort from lower EL (Page Fault / CoW!)
    uint64_t far;
    __asm__ volatile("mrs %0, far_el1" : "=r"(far));
    if (handle_page_fault(current_task()->mm, far, esr) != 0) {
      sys_exit_task(-SIGSEGV);
    }
  } else {
    sys_exit_task(-SIGILL);
  }
}
Makefile — Cross-Compile & Automated QEMU Test Runner 
CC = aarch64-elf-gcc
CFLAGS = -Wall -Wextra -Werror -ffreestanding -nostdlib -mgeneral-regs-only -O2 -Ikernel
LDFLAGS = -T linker.ld -nostdlib

OBJS = build/boot.o build/vectors.o build/switch.o build/main.o build/trap.o build/sched.o build/mm_mmu.o build/ipc_binder.o

build/aegis.elf: $(OBJS)
	$(CC) $(LDFLAGS) -o $@ $^

run: build/aegis.elf
	qemu-system-aarch64 -M virt -cpu cortex-a72 -nographic -m 256M -kernel $<

test: build/aegis.elf
	python3 tests/run_qemu_tests.py build/aegis.elf
