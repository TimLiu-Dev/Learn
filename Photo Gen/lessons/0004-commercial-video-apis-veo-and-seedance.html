2. Video APIs Are Always Asynchronous Polling Loops

Android Analogy: Video generation takes 30–180 seconds. Calling a blocking HTTP request would time out immediately—treat it like enqueuing a WorkRequest and polling with a timeout + backoff until operation.done. Notice the two-phase pattern: submit job → receive operation handle → poll with backoff/timeout → inspect operation.error → download MP4 bytes.

Python · Google Veo 3.1 Ingredients-to-Video (Valid GenerateVideosConfig)
import time
from google import genai
from google.genai import types

client = genai.Client()

# Up to 3 reference images ("Ingredients-to-Video") lock coat markings from multiple angles
ref_front = types.VideoGenerationReferenceImage(
    image=types.Image.from_file(location="datasets/cat_raw/cat_face_01.jpg"),
    reference_type="asset",
)
ref_side = types.VideoGenerationReferenceImage(
    image=types.Image.from_file(location="datasets/cat_raw/cat_side_02.jpg"),
    reference_type="asset",
)

operation = client.models.generate_videos(
    model="veo-3.1-generate-001",
    prompt=(
        "The ginger tabby cat blinks slowly in golden morning window light, "
        "stretches its front paws, and bats playfully at a dust mote. "
        "Soft purring and subtle sunlit room ambience."
    ),
    config=types.GenerateVideosConfig(
        aspect_ratio="9:16",
        resolution="1080p",
        duration_seconds=8,
        number_of_videos=1,
        reference_images=[ref_front, ref_side],
    ),
)

deadline = time.monotonic() + 300
while not operation.done:
    if time.monotonic() > deadline:
        raise TimeoutError("Veo generation exceeded 5-minute timeout")
    time.sleep(15)
    operation = client.operations.get(operation)

if operation.error:
    raise RuntimeError(f"Veo failed: {operation.error}")

video_file = operation.response.generated_videos[0].video
client.files.download(file=video_file)
video_file.save("cat_prototype_veo.mp4")
Python · Runnable Seedance 2.5 via fal_client Gateway
import fal_client  # pip install fal-client; uses FAL_KEY env var

# Pair static cat photos (coat markings) + a real 3s cat walking clip (motion physics)
result = fal_client.subscribe(
    "fal-ai/bytedance/seedance/v2.5/reference-to-video",
    arguments={
        "prompt": "Multi-shot sequence: the exact cat walks across a wooden floor then leaps onto a sunlit sofa.",
        "duration": "10",
        "resolution": "1080p",
        "aspect_ratio": "9:16",
        "image_urls": [
            fal_client.upload_file("datasets/cat_raw/cat_face_01.jpg"),
            fal_client.upload_file("datasets/cat_raw/cat_side_02.jpg"),
        ],
        "video_urls": [
            fal_client.upload_file("datasets/cat_video_refs/cat_walk_3s.mp4"),
        ],
    },
    with_logs=True,
)
print("Download URL:", result["video"]["url"])
