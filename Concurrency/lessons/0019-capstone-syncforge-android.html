1. OutboxDao.kt & SyncCoordinator.kt (Single-Flight + WAL Outbox)

data class OutboxEntry(
    val id: String,
    val entityId: String,
    val idempotencyKey: String,
    val baseVersion: Long,
    val payloadJson: String,
    val status: OutboxStatus // PENDING, IN_FLIGHT, SENT, FAILED
)

class SyncCoordinator(
    private val outboxDao: OutboxDao,
    private val api: SyncApi,
    private val ioDispatcher: CoroutineDispatcher,
    private val conflictResolver: ConflictResolver
) {
    private val inFlightSync = Mutex()
    private var activeSyncJob: Deferred<SyncSummary>? = null

    // Single-flight: concurrent UI pull-to-refresh + WorkManager share one run
    suspend fun syncNow(): SyncSummary = coroutineScope {
        val deferred = inFlightSync.withLock {
            activeSyncJob?.takeIf { it.isActive } ?: async(ioDispatcher) {
                executeDrain()
            }.also { activeSyncJob = it }
        }
        deferred.await()
    }

    private suspend fun executeDrain(): SyncSummary {
        outboxDao.recoverStaleInFlight() // Crash recovery: IN_FLIGHT -> PENDING
        val pending = outboxDao.claimBatch()
        var synced = 0
        for (entry in pending) {
            currentCoroutineContext().ensureActive()
            when (val res = api.pushMutation(entry, entry.idempotencyKey)) {
                is ApiResult.Ok -> { outboxDao.markSent(entry.id); synced++ }
                is ApiResult.Conflict409 -> {
                    val merged = conflictResolver.merge(entry.payloadJson, res.serverSnapshot)
                    outboxDao.requeueMerged(entry.id, merged, res.serverVersion)
                }
                is ApiResult.TransientError -> outboxDao.releaseToPending(entry.id)
            }
        }
        return SyncSummary(syncedCount = synced)
    }
}

2. MediaUploadPipeline.kt & SyncCoordinatorTest.kt

class MediaUploadPipeline(private val api: MediaApi) {
    private val fileSemaphore = Semaphore(permits = 2)

    suspend fun uploadAll(files: List<MediaJob>, onProgress: (String, Int) -> Unit) = coroutineScope {
        files.map { file ->
            async {
                fileSemaphore.withPermit {
                    val chunkSem = Semaphore(permits = 3)
                    val remaining = file.chunks.drop(file.ackedOffset)
                    remaining.map { chunk ->
                        async {
                            chunkSem.withPermit {
                                api.uploadChunk(file.id, chunk, file.idempotencyKey)
                            }
                        }
                    }.awaitAll()
                    onProgress(file.id, 100)
                }
            }
        }.awaitAll()
    }
}

SyncForgeOutboxEngine

Implement SyncForgeOutboxEngine supporting crash recovery of stale IN_FLIGHT entries, idempotency keys, and 409 version conflict merge & retry.

Not Run
• enqueue(entityId, payload, baseVersion, idempotencyKey): deduplicates by idempotencyKey if already queued/sent; stores { id, entityId, payload, baseVersion, idempotencyKey, status: 'PENDING' }.
• recoverCrashedInFlight(): transitions any 'IN_FLIGHT' rows back to 'PENDING' (simulating process restart recovery).
• async drain(pushFn, mergeFn): processes 'PENDING' entries sequentially (`IN_FLIGHT ->` call `await pushFn(entry)`).

  - If res.status === 'CONFLICT_409': merge with mergeFn(entry.payload, res.serverPayload), update baseVersion = res.serverVersion, retry immediately once via `pushFn`; if that returns `'OK'`, mark `'SENT'`, else set `'PENDING'`.
  - If pushFn throws transient error: reset status back to 'PENDING' so next drain retries safely. Return `{ sent, pending }`.
