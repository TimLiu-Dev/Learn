The Golden Rule of Progressive Rounds: Never hardcode raw primitives (Map<String, String>) in Stage 1 if timestamps, TTLs, or versions may arrive in Stage 2! Wrap values in a lightweight Entry(value, expiresAt) record and route all reads through a single helper liveEntry(key, now) so Stages 2 and 3 extend cleanly in 10 lines instead of forcing a panicked rewrite.

class ProgressiveStore {

    private val backups = HashMap<Int, Map<String, Entry>>()
    private var nextId = 1

from dataclasses import dataclass
from typing import Optional

@dataclass(frozen=True)
class Entry:
    value: str
    expires_at: Optional[int] = None

    def is_alive(self, now: int) -> bool:
        return self.expires_at is None or now < self.expires_at

class ProgressiveStore:
    def __init__(self):
        self.store: dict[str, Entry] = {}
        self.backups: dict[int, dict[str, Entry]] = {}
        self.next_id = 1

    def _live(self, key: str, now: int) -> Optional[Entry]:
        ent = self.store.get(key)
        return ent if (ent and ent.is_alive(now)) else None

    def set_at(self, key: str, value: str, now: int, ttl: Optional[int] = None) -> None:
        exp = (now + ttl) if ttl is not None else None
        self.store[key] = Entry(value, exp)

    def get_at(self, key: str, now: int) -> Optional[str]:
        ent = self._live(key, now)
        return ent.value if ent else None

    def backup(self, now: int) -> int:
        bid = self.next_id; self.next_id += 1
        self.backups[bid] = {k: v for k, v in self.store.items() if v.is_alive(now)}
        return bid

class ProgressiveStore {
    record Entry(String value, Long expiresAt) {
        boolean isAlive(long now) { return expiresAt == null || now < expiresAt; }
    }

    private Map<String, Entry> store = new HashMap<>();
    private final Map<Integer, Map<String, Entry>> backups = new HashMap<>();
    private int nextId = 1;

    private Entry live(String key, long now) {
        Entry e = store.get(key);
        return (e != null && e.isAlive(now)) ? e : null;
    }

    public void setAt(String key, String value, long now, Long ttl) {
        store.put(key, new Entry(value, ttl == null ? null : now + ttl));
    }

    public String getAt(String key, long now) {
        Entry e = live(key, now);
        return e == null ? null : e.value();
    }

    public int backup(long now) {
        Map<String, Entry> snap = new HashMap<>();
        store.forEach((k, v) -> { if (v.isAlive(now)) snap.put(k, v); });
        backups.put(nextId, snap);
        return nextId++;
    }
}

Walkthrough: Progressive 3-Stage In-Memory KV Store

Build an in-memory database supporting: Stage 1: set(k, v), get(k), delete(k); Stage 2: setAt(k, v, ts, ttl), getAt(k, ts) (half-open TTL [ts, ts + ttl)), and sorted scanPrefixAt(prefix, ts); Stage 3: backup(ts) and restore(ts, backupId) (recalculating remaining TTL relative to restore time!).

1 · Clarify

Is TTL boundary inclusive or exclusive at ts + ttl? Standard spec is half-open: alive strictly while now < ts + ttl (expired at now == ts + ttl). When restoring a backup taken at backupTs at later time restoreTs, does remaining TTL carry over? Yes! Store remaining = expiresAt - backupTs inside the backup so restored expiration becomes restoreTs + remaining!

2 · Examples

setAt("a:1", "v1", ts=10, ttl=5) (expires at 15). backup(ts=12) → remaining TTL is 15 - 12 = 3. getAt("a:1", ts=15) → null (expired). restore(ts=20, backupId=1) → new expiry 20 + 3 = 23; getAt("a:1", ts=22) → "v1"!

3 · Approach

1. Represent every record with immutable Entry(value, expiresAt) (expiresAt = null for Stage 1 non-expiring keys).

2. Delegate Stage 1 set/get/delete directly to Stage 2 methods with ts = 0, ttl = null—zero code duplication.

3. On backup(ts), snapshot live entries storing SavedEntry(value, remainingTtl); on restore(ts, id), rebuild store setting expiresAt = ts + remainingTtl.

class CloudKVStore {
    private data class Entry(val value: String, val expiresAt: Long?) {
        fun isAlive(now: Long) = expiresAt == null || now < expiresAt
    }
    private data class Saved(val value: String, val remainingTtl: Long?)

    private var store = HashMap<String, Entry>()
    private val backups = HashMap<Int, Map<String, Saved>>()
    private var nextId = 1

    fun set(key: String, value: String) = setAt(key, value, 0L, null)
    fun get(key: String): String? = getAt(key, 0L)
    fun delete(key: String): Boolean = deleteAt(key, 0L)

    fun setAt(key: String, value: String, ts: Long, ttl: Long? = null) {
        store[key] = Entry(value, ttl?.let { ts + it })
    }

    fun getAt(key: String, ts: Long): String? =
        store[key]?.takeIf { it.isAlive(ts) }?.value

    fun deleteAt(key: String, ts: Long): Boolean {
        val alive = store[key]?.isAlive(ts) == true
        store.remove(key)
        return alive
    }

    fun scanPrefixAt(prefix: String, ts: Long): List<String> =
        store.entries
            .filter { (k, v) -> k.startsWith(prefix) && v.isAlive(ts) }
            .sortedBy { it.key }
            .map { "${it.key}(${it.value.value})" }

    fun backup(ts: Long): Int {
        val snap = HashMap<String, Saved>()
        for ((k, v) in store) {
            if (v.isAlive(ts)) {
                snap[k] = Saved(v.value, v.expiresAt?.let { it - ts })
            }
        }
        backups[nextId] = snap
        return nextId++
    }

    fun restore(ts: Long, backupId: Int) {
        val snap = backups[backupId] ?: return
        val nextStore = HashMap<String, Entry>()
        for ((k, s) in snap) {
            nextStore[k] = Entry(s.value, s.remainingTtl?.let { ts + it })
        }
        store = nextStore
    }
}
class CloudKVStore:
    def __init__(self):
        self.store = {}   # key -> (value, expires_at or None)
        self.backups = {} # id -> dict[key, (value, remaining_ttl or None)]
        self.next_id = 1

    def _alive(self, key: str, ts: int) -> bool:
        if key not in self.store:
            return False
        _, exp = self.store[key]
        return exp is None or ts < exp

    def set(self, key: str, value: str) -> None:
        self.set_at(key, value, 0, None)

    def get(self, key: str):
        return self.get_at(key, 0)

    def set_at(self, key: str, value: str, ts: int, ttl: int = None) -> None:
        self.store[key] = (value, ts + ttl if ttl is not None else None)

    def get_at(self, key: str, ts: int):
        return self.store[key][0] if self._alive(key, ts) else None

    def delete_at(self, key: str, ts: int) -> bool:
        was_alive = self._alive(key, ts)
        self.store.pop(key, None)
        return was_alive

    def scan_prefix_at(self, prefix: str, ts: int) -> list[str]:
        items = [
            f"{k}({v})" for k, (v, exp) in self.store.items()
            if k.startswith(prefix) and (exp is None or ts < exp)
        ]
        items.sort()
        return items

    def backup(self, ts: int) -> int:
        bid = self.next_id; self.next_id += 1
        self.backups[bid] = {
            k: (v, (exp - ts) if exp is not None else None)
            for k, (v, exp) in self.store.items()
            if exp is None or ts < exp
        }
        return bid

    def restore(self, ts: int, backup_id: int) -> None:
        if backup_id not in self.backups:
            return
        self.store = {
            k: (v, (ts + rem) if rem is not None else None)
            for k, (v, rem) in self.backups[backup_id].items()
        }
class CloudKVStore {
    private record Entry(String value, Long expiresAt) {
        boolean isAlive(long ts) { return expiresAt == null || ts < expiresAt; }
    }
    private record Saved(String value, Long remainingTtl) {}

    private Map<String, Entry> store = new HashMap<>();
    private final Map<Integer, Map<String, Saved>> backups = new HashMap<>();
    private int nextId = 1;

    public void set(String key, String value) { setAt(key, value, 0L, null); }
    public String get(String key) { return getAt(key, 0L); }

    public void setAt(String key, String value, long ts, Long ttl) {
        store.put(key, new Entry(value, ttl == null ? null : ts + ttl));
    }

    public String getAt(String key, long ts) {
        Entry e = store.get(key);
        return (e != null && e.isAlive(ts)) ? e.value() : null;
    }

    public boolean deleteAt(String key, long ts) {
        Entry e = store.remove(key);
        return e != null && e.isAlive(ts);
    }

    public List<String> scanPrefixAt(String prefix, long ts) {
        List<String> keys = new ArrayList<>();
        for (var entry : store.entrySet()) {
            if (entry.getKey().startsWith(prefix) && entry.getValue().isAlive(ts)) {
                keys.add(entry.getKey());
            }
        }
        Collections.sort(keys);
        List<String> res = new ArrayList<>(keys.size());
        for (String k : keys) res.add(k + "(" + store.get(k).value() + ")");
        return res;
    }

    public int backup(long ts) {
        Map<String, Saved> snap = new HashMap<>();
        for (var entry : store.entrySet()) {
            Entry v = entry.getValue();
            if (v.isAlive(ts)) {
                snap.put(entry.getKey(), new Saved(v.value(), v.expiresAt() == null ? null : v.expiresAt() - ts));
            }
        }
        backups.put(nextId, snap);
        return nextId++;
    }

    public void restore(long ts, int backupId) {
        Map<String, Saved> snap = backups.get(backupId);
        if (snap == null) return;
        Map<String, Entry> next = new HashMap<>();
        for (var entry : snap.entrySet()) {
            Saved s = entry.getValue();
            next.put(entry.getKey(), new Entry(s.value(), s.remainingTtl() == null ? null : ts + s.remainingTtl()));
        }
        store = next;
    }
}

Teardown 1 (JVM / Kotlin & Java): Spot 3 Bugs in Leaderboard Cache

class Leaderboard {
    Map<Integer, Integer> scores = new HashMap<>();
    PriorityQueue<int[]> pq = new PriorityQueue<>((a, b) -> b[1] - a[1]);

    public void record(int userId, int delta) {
        scores.put(userId, scores.getOrDefault(userId, 0) + delta);
        pq.offer(new int[]{userId, scores.get(userId)});
    }
    public boolean hasSameTopScore(int u1, int u2) {
        return scores.get(u1) == scores.get(u2);
    }
}
  1. Bug 1 (Boxed Integer == Trap): scores.get(u1) == scores.get(u2) compares Integer object references! Scores outside -128..127 return false even when equal (and throws NPE if missing)—use Objects.equals(scores.get(u1), scores.get(u2)).
  2. Bug 2 (Comparator Overflow): (a, b) -> b[1] - a[1] overflows 32-bit int when scores have opposite signs—use Integer.compare(b[1], a[1]).
  3. Bug 3 (Stale Duplicate Heap Entries): record pushes a new int[] on every update without discarding stale entries; polling top-K will return the same userId multiple times or stale scores unless you skip popped entries where cur[1] != scores.get(cur[0])!

Teardown 2 (Python): Spot 3 Bugs in Shortest Grid Path

def shortest_path(grid: list[list[int]]) -> int:
    R, C = len(grid), len(grid[0])
    visited = [[False] * C] * R
    q = [(0, 0, 0)]
    while q:
        r, c, d = q.pop(0)
        if (r, c) == (R - 1, C - 1): return d
        visited[r][c] = True
        for dr, dc in ((1,0), (-1,0), (0,1), (0,-1)):
            nr, nc = r + dr, c + dc
            if 0 <= nr < R and 0 <= nc < C and not visited[nr][nc] and grid[nr][nc] == 0:
                q.append((nr, nc, d + 1))
    return -1
  1. Bug 1 (Shallow Row Aliasing): [[False] * C] * R aliases the exact same row reference R times—setting visited[0][0] = True marks column 0 visited in every row! Use [[False] * C for _ in range(R)].
  2. Bug 2 (Late Visited Marking → Memory Blowup): Marking visited[r][c] = True on pop instead of enqueue lets many frontier cells push the same neighbor repeatedly (up to exponential/quadratic duplicates) and skips checking grid[0][0] == 1!
  3. Bug 3 (Hidden O((RC)²) Queue): q.pop(0) on a Python list shifts all elements in O(k)—use collections.deque.popleft().

Teardown 3 (Kotlin / Java Concurrency): Spot 3 Bugs in Thread-Safe LRU

class ConcurrentLRU(private val cap: Int) {
    private val lock = java.util.concurrent.locks.ReentrantReadWriteLock()
    private val map = object : LinkedHashMap<Int, Int>(cap, 0.75f, true) {
        override fun removeEldestEntry(e: MutableMap.MutableEntry<Int, Int>?) = size > cap
    }
    fun get(k: Int): Int? {
        lock.readLock().lock()
        return map[k].also { lock.readLock().unlock() }
    }
}
  1. Bug 1 (Data Race Under ReadLock!): LinkedHashMap(..., accessOrder = true) structurally mutates its internal doubly linked list on every map[k] read! Concurrent threads holding readLock() corrupt pointers and infinite-loop—get() needs an exclusive lock (or striped/sharded locks).
  2. Bug 2 (Missing try / finally): If map[k] throws an exception, unlock() never runs, deadlocking the process permanently—always use lock.writeLock().withLock { ... } or try / finally.
  3. Bug 3 (Contention Bottleneck Follow-up): A single global write lock serializes all threads; in production, shard keys across K = 16 independent locked segments via (k.hashCode() and 0x7fffffff) % K.
