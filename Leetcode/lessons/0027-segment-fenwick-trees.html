Dynamic range sum / min queries and point updates in O(log n) time.

Warm-up from memory

Close this page and recall: why does a static Prefix Sum array fail when there are frequent updates to individual array elements?

The idea

A Prefix Sum array gives O(1) range queries, but updating an element takes O(n). When an array experiences BOTH frequent point/range updates and range sum/min queries, Segment Trees and Fenwick Trees (Binary Indexed Trees) achieve O(log n) for both update and query. A Fenwick Tree uses bit tricks (i & (-i)) for minimal code.

Reach for it when you see…

  • 'Range sum query with mutable / update operations'
  • 'Count of smaller numbers after self' (inversion counting)
  • Dynamic range minimum/maximum with updates

Pattern templates

class FenwickTree(val n: Int) {
    val tree = IntArray(n + 1)
    fun update(index: Int, delta: Int) {
        var i = index + 1
        while (i <= n) { tree[i] += delta; i += i and -i }
    }
    fun query(index: Int): Int {
        var sum = 0; var i = index + 1
        while (i > 0) { sum += tree[i]; i -= i and -i }
        return sum
    }
}
class FenwickTree:
    def __init__(self, n: int):
        self.tree = [0] * (n + 1)
    def update(self, index: int, delta: int):
        i = index + 1
        while i < len(self.tree):
            self.tree[i] += delta
            i += i & (-i)
    def query(self, index: int) -> int:
        s, i = 0, index + 1
        while i > 0:
            s += self.tree[i]
            i -= i & (-i)
        return s
class FenwickTree {
    int[] tree;
    public FenwickTree(int n) { tree = new int[n + 1]; }
    public void update(int index, int delta) {
        for (int i = index + 1; i < tree.length; i += i & -i) tree[i] += delta;
    }
    public int query(int index) {
        int sum = 0;
        for (int i = index + 1; i > 0; i -= i & -i) sum += tree[i];
        return sum;
    }
}

Walkthrough: Range Sum Query - Mutable

Given an integer array nums, handle two operations: update(index, val) to update nums[index], and sumRange(left, right) to return sum between left and right inclusive.

1 · Clarify

How many queries and updates? Up to 3 * 10^4 operations. Numbers can be negative.

2 · Examples

nums = [1, 3, 5]. sumRange(0, 2) → 9. update(1, 2) → nums = [1, 2, 5]. sumRange(0, 2) → 8.

3 · Approach

1. A Fenwick Tree (Binary Indexed Tree) stores partial prefix sums in an array of size n + 1.

2. i += i & -i moves to parent covering a wider range during updates.

3. i -= i & -i removes the lowest set bit to accumulate prefix sums.

4. sumRange(left, right) = query(right) - query(left - 1) in O(log n).

4 · Code 
class NumArray(private val nums: IntArray) {
    private val n = nums.size
    private val tree = IntArray(n + 1)

    init {
        for (i in nums.indices) add(i + 1, nums[i])
    }

    private fun add(index: Int, delta: Int) {
        var i = index
        while (i <= n) {
            tree[i] += delta
            i += i and -i
        }
    }

    fun update(index: Int, `val`: Int) {
        val delta = `val` - nums[index]
        nums[index] = `val`
        add(index + 1, delta)
    }

    private fun query(index: Int): Int {
        var sum = 0
        var i = index
        while (i > 0) {
            sum += tree[i]
            i -= i and -i
        }
        return sum
    }

    fun sumRange(left: Int, right: Int): Int = query(right + 1) - query(left)
}
class NumArray:
    def __init__(self, nums: list[int]):
        self.nums = nums
        self.n = len(nums)
        self.tree = [0] * (self.n + 1)
        for i, x in enumerate(nums):
            self._add(i + 1, x)

    def _add(self, i: int, delta: int):
        while i <= self.n:
            self.tree[i] += delta
            i += i & (-i)

    def update(self, index: int, val: int) -> None:
        delta = val - self.nums[index]
        self.nums[index] = val
        self._add(index + 1, delta)

    def _query(self, i: int) -> int:
        s = 0
        while i > 0:
            s += self.tree[i]
            i -= i & (-i)
        return s

    def sumRange(self, left: int, right: int) -> int:
        return self._query(right + 1) - self._query(left)
class NumArray {
    private final int[] nums;
    private final int[] tree;
    private final int n;

    public NumArray(int[] nums) {
        this.nums = nums;
        this.n = nums.length;
        this.tree = new int[n + 1];
        for (int i = 0; i < n; i++) add(i + 1, nums[i]);
    }

    private void add(int i, int delta) {
        while (i <= n) {
            tree[i] += delta;
            i += i & -i;
        }
    }

    public void update(int index, int val) {
        int delta = val - nums[index];
        nums[index] = val;
        add(index + 1, delta);
    }

    private int query(int i) {
        int sum = 0;
        while (i > 0) {
            sum += tree[i];
            i -= i & -i;
        }
        return sum;
    }

    public int sumRange(int left, int right) {
        return query(right + 1) - query(left);
    }
}

class NumArray {
    private int[] tree, nums;
    private int n;
    public NumArray(int[] nums) {
        this.nums = nums.clone();
        this.n = nums.length;
        this.tree = new int[n + 1];
        for (int i = 0; i < n; i++) add(i + 1, nums[i]);
    }
    private void add(int i, int delta) {
        for (; i <= n; i += i & -i) tree[i] += delta;
    }
    private int query(int i) {
        int s = 0;
        for (; i > 0; i -= i & -i) s += tree[i];
        return s;
    }
    public void update(int index, int val) {
        int delta = val - nums[index];
        nums[index] = val;
        add(index + 1, delta);
    }
    public int sumRange(int left, int right) {
        return query(right + 1) - query(left);
    }
}

Quick check

What does the lowbit expression i & (-i) isolate in binary arithmetic?

  1. Highest bit in binary sequence
  2. Count of active ones in integer
  3. Twos complement sign bit value
  4. Lowest set power-of-two bit val

i & (-i) extracts the lowest 1-bit in i (e.g. 12 = 1100_2 &rarr; 12 & -12 = 4 = 0100_2).

When is a Segment Tree strictly necessary over a Fenwick Tree?

  1. Arbitrary range min or max queries
  2. Running simple prefix sum counts
  3. Array size is strictly power of 2
  4. Graph nodes need cycle checks

Fenwick Trees rely on prefix subtractions query(r) - query(l-1), which works for sums, but min/max cannot be subtracted.

What is the memory size required for a standard recursive Segment Tree on array of size n?

  1. 2 * n elements allocated array
  2. 4 * n elements allocated array
  3. n * log n total tree node list
  4. n^2 matrix representation space

A binary segment tree array requires up to 4n space to safely hold all leaf and internal nodes.

  1. Coordinate-compress values into ranks 1..M.
  2. Scan array backwards from right to left.
  3. For each number, query count in Fenwick Tree for rank < current, then insert current number (update(rank, 1)).

Queue Reconstruction by Height Medium

  1. Sort people: height descending, k ascending.
  2. Insert each person at index k in a list, or use a Segment Tree to find the k-th empty slot in O(log n).

Range Sum Query 2D - Mutable Hard

  1. Use a 2D Fenwick Tree (nested loops with i += i & -i and j += j & -j).
  2. Update and query both run in O(log M * log N) time.
