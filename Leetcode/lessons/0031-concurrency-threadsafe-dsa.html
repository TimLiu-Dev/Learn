Walkthrough: Design Bounded Blocking Queue

Implement a thread-safe BoundedBlockingQueue: enqueue(element) blocks if queue is full until space is available; dequeue() blocks if queue is empty until an element is added.

1 · Clarify

Can multiple producers and consumers call concurrently? What exception should be thrown on thread interruption? Multiple concurrent threads. Use while loop (not if) around await to prevent spurious wakeups.

2 · Examples

Queue cap = 2. Producer pushes 1, 2. Third push blocks. Consumer pops 1 → unblocks producer to push 3.

3 · Approach

1. Classical Producer-Consumer pattern.

2. Guard queue state with ReentrantLock.

3. Two Condition variables: notFull (producers wait when count == capacity) and notEmpty (consumers wait when count == 0).

4. Always wrap await in a while loop to defend against spurious wakeups.

4 · Code 
import java.util.concurrent.locks.ReentrantLock

class BoundedBlockingQueue(private val capacity: Int) {
    private val queue = IntArray(capacity)
    private var head = 0; private var tail = 0; private var count = 0
    private val lock = ReentrantLock()
    private val notFull = lock.newCondition()
    private val notEmpty = lock.newCondition()

    fun enqueue(element: Int) {
        lock.lock()
        try {
            while (count == capacity) {
                notFull.await()
            }
            queue[tail] = element
            tail = (tail + 1) % capacity
            count++
            notEmpty.signal()
        } finally {
            lock.unlock()
        }
    }

    fun dequeue(): Int {
        lock.lock()
        try {
            while (count == 0) {
                notEmpty.await()
            }
            val element = queue[head]
            head = (head + 1) % capacity
            count--
            notFull.signal()
            return element
        } finally {
            lock.unlock()
        }
    }

    fun size(): Int {
        lock.lock()
        try { return count } finally { lock.unlock() }
    }
}
import threading

class BoundedBlockingQueue:
    def __init__(self, capacity: int):
        self.capacity = capacity
        self.queue = []
        self.lock = threading.Lock()
        self.not_full = threading.Condition(self.lock)
        self.not_empty = threading.Condition(self.lock)

    def enqueue(self, element: int) -> None:
        with self.not_full:
            while len(self.queue) == self.capacity:
                self.not_full.wait()
            self.queue.append(element)
            self.not_empty.notify()

    def dequeue(self) -> int:
        with self.not_empty:
            while len(self.queue) == 0:
                self.not_empty.wait()
            element = self.queue.pop(0)
            self.not_full.notify()
            return element

    def size(self) -> int:
        with self.lock:
            return len(self.queue)
import java.util.concurrent.locks.Condition;
import java.util.concurrent.locks.ReentrantLock;

class BoundedBlockingQueue {
    private final int[] queue;
    private final int capacity;
    private int head = 0, tail = 0, count = 0;
    private final ReentrantLock lock = new ReentrantLock();
    private final Condition notFull = lock.newCondition();
    private final Condition notEmpty = lock.newCondition();

    public BoundedBlockingQueue(int capacity) {
        this.capacity = capacity;
        this.queue = new int[capacity];
    }

    public void enqueue(int element) throws InterruptedException {
        lock.lock();
        try {
            while (count == capacity) {
                notFull.await();
            }
            queue[tail] = element;
            tail = (tail + 1) % capacity;
            count++;
            notEmpty.signal();
        } finally {
            lock.unlock();
        }
    }

    public int dequeue() throws InterruptedException {
        lock.lock();
        try {
            while (count == 0) {
                notEmpty.await();
            }
            int element = queue[head];
            head = (head + 1) % capacity;
            count--;
            notFull.signal();
            return element;
        } finally {
            lock.unlock();
        }
    }

    public int size() {
        lock.lock();
        try {
            return count;
        } finally {
            lock.unlock();
        }
    }
}

    private val queue = ArrayDeque<Int>()
    private val lock = java.util.concurrent.locks.ReentrantLock()
    private val notFull = lock.newCondition()
    private val notEmpty = lock.newCondition()
    fun enqueue(element: Int) {

        try {
            while (queue.size == capacity) notFull.await()
            queue.addLast(element)
            notEmpty.signal()
        } finally { lock.unlock() }
    }
    fun dequeue(): Int {
        lock.lock()
        try {
            while (queue.isEmpty()) notEmpty.await()
            val item = queue.removeFirst()
            notFull.signal()
            return item
        } finally { lock.unlock() }
    }
    fun size(): Int {
        lock.lock()
        try { return queue.size } finally { lock.unlock() }
    }
}

            return item

            return len(self.queue)
class BoundedBlockingQueue {
    private final int capacity;
    private final Deque<Integer> queue = new ArrayDeque<>();
    private final ReentrantLock lock = new ReentrantLock();
    private final Condition notFull = lock.newCondition();
    private final Condition notEmpty = lock.newCondition();
    public BoundedBlockingQueue(int capacity) { this.capacity = capacity; }
    public void enqueue(int element) throws InterruptedException {
        lock.lock();
        try {
            while (queue.size() == capacity) notFull.await();
            queue.addLast(element);
            notEmpty.signal();
        } finally { lock.unlock(); }
    }
    public int dequeue() throws InterruptedException {
        lock.lock();
        try {
            while (queue.isEmpty()) notEmpty.await();
            int val = queue.removeFirst();
            notFull.signal();
            return val;
        } finally { lock.unlock(); }
    }
    public int size() {
        lock.lock();
        try { return queue.size(); } finally { lock.unlock(); }
    }
}

Quick check

Why MUST condition.await() / wait() be wrapped in a while loop rather than an if statement?

  1. Releases locks during recursion
  2. Increases thread context speed
  3. Sorts waiting thread priorities
  4. Guards against spurious wakeups

A thread can wake up spuriously without a signal, or another thread might consume the item first; re-checking the condition is mandatory.

Why is lock.unlock() ALWAYS placed inside a finally block in Java and Kotlin?

  1. Ensures lock release on exception
  2. Accelerates heap garbage collector
  3. Flushes volatile CPU registers
  4. Prevents memory segmentation faults

If an exception is thrown inside the critical section, failing to release the lock in a finally block causes permanent deadlocks.

How does ReadWriteLock optimize read-heavy data structures compared to a mutual exclusion lock?

  1. Eliminates all write block locks
  2. Allows concurrent reads by threads
  3. Sorts read threads by arrival time
  4. Uses zero kernel context switches

Multiple reader threads can hold the read lock concurrently; only writers require exclusive access.
