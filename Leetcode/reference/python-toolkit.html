Lists & Arrays

# Fixed size list initialization
arr = [0] * n                         # 1D array of zeros
matrix = [[0] * cols for _ in range(rows)]  # NEVER [[0]*cols]*rows (shallow copy trap!)

# Basic operations
arr.append(5); arr.pop()              # O(1) push/pop at end
arr.sort()                            # In-place Timsort O(n log n)
sorted_arr = sorted(arr)              # Returns new list
n = len(arr); last = arr[-1]

# Slicing: creates a NEW copy! Takes O(k) time and space
sub = arr[1:4]                        # [1, 4)
rev = arr[::-1]                       # reversed copy

# Iteration
for i in range(len(arr)): ...
for x in arr: ...
for i, x in enumerate(arr): ...
for a, b in zip(list1, list2): ...

Strings & Characters

s = "hello"
ch = s[0]; length = len(s)
sub = s[1:3]                          # "el"

# Character arithmetic (ASCII offset)
idx = ord(ch) - ord('a')              # 0..25 for lowercase letters
char_from_idx = chr(ord('a') + idx)   # 'c'
ch.isalnum(); ch.isalpha(); ch.isdigit(); ch.lower()

# Building strings: NEVER += in a loop (O(n²) time due to immutability)
chars = ['a', 'b', 'c']
result = "".join(chars)               # O(n) construction

Dictionaries & Sets

# Standard dictionary
m = {}
m["key"] = 1
val = m.get("missing", 0)             # default if absent
if "key" in m: del m["key"]

# defaultdict and Counter
from collections import defaultdict, Counter
counts = defaultdict(int)             # auto-initializes missing keys to 0
for ch in s: counts[ch] += 1

grouped = defaultdict(list)
grouped["prime"].append(7)

freq = Counter(arr)                   # counts frequencies in O(n)
top2 = freq.most_common(2)            # [(val, count), ...]

# Set operations: average O(1) lookup
seen = set()
seen.add(10); seen.discard(10)        # discard does not throw if absent

Stack, Queue, Deque

# Stack: use list
stack = []
stack.append(1); top = stack[-1]; stack.pop()
is_empty = len(stack) == 0

# Queue & Deque: ALWAYS use collections.deque!
from collections import deque
q = deque([1, 2, 3])
q.append(4)                           # push right
left = q.popleft()                    # O(1) pop from front (list.pop(0) is O(n)!)
q.appendleft(0)                       # O(1) push left
right = q.pop()                       # O(1) pop right

Heap (Priority Queue)

import heapq

# Min-Heap by default
h = []
heapq.heappush(h, 10)
heapq.heappush(h, 3)
smallest = heapq.heappop(h)           # 3
peek = h[0]

# Heapify in-place in O(n)
nums = [5, 1, 9, 3]
heapq.heapify(nums)

# Max-Heap trick: negate values
max_h = []
heapq.heappush(max_h, -10)
largest = -heapq.heappop(max_h)       # 10

# Storing tuples (priority, value)
heapq.heappush(h, (distance, node_id))

Gotchas & 2027 Big Tech Nuances

  • List initialization shallow copy: [[0]*n]*m duplicates the same list reference $m$ times. Modifying row 0 modifies all rows! Always write [[0]*n for _ in range(m)].
  • list.pop(0) is $O(n)$ disaster: Calling pop(0) in a loop turns an $O(n)$ BFS into $O(n^2)$. Always use collections.deque.popleft().
  • Arbitrary precision integers: Python automatically promotes large integers without overflow. However, division / returns a float; use integer division //.
  • Negative integer division: In Python, -7 // 2 == -4 (floor toward $-\infty$), whereas in Java and Kotlin -7 / 2 == -3 (truncate toward zero). For division toward zero, use int(-7 / 2).
  • Recursion limit: Default recursion depth is 1,000. In deep tree/graph DFS, call import sys; sys.setrecursionlimit(200_000) or switch to iterative DFS.
